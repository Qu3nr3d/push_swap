Normal tests

Works with:
* no arguments
./push_swap | ./checker_linux
valgrind ./push_swap | ./checker_linux
* one number
./push_swap 5 | ./checker_linux 5
valgrind ./push_swap 5 | ./checker_linux 5
* two numbers
./push_swap 12 5 | ./checker_linux 12 5
valgrind ./push_swap 12 5 | ./checker_linux 12 5
* sorted array for each algorithm
./push_swap --simple 45 67 90 900 | ./checker_linux 45 67 90 900
valgrind ./push_swap --simple 45 67 90 900 | ./checker_linux 45 67 90 900
./push_swap --medium 45 67 90 900 | ./checker_linux 45 67 90 900
valgrind ./push_swap --medium 45 67 90 900 | ./checker_linux 45 67 90 900
./push_swap --complex 45 67 90 900 | ./checker_linux 45 67 90 900
valgrind ./push_swap --complex 45 67 90 900 | ./checker_linux 45 67 90 900
* bench
./push_swap --bench 12 5 --complex | ./checker_linux 12 5
valgrind ./push_swap --bench 12 5 --complex | ./checker_linux 12 5
./push_swap --bench --simple 12 5 | ./checker_linux 12 5
valgrind ./push_swap --bench --simple 12 5 | ./checker_linux 12 5
* checks if is error
./push_swap 12 45 3 34 3 | ./checker_linux 12 45 3 34 3
valgrind ./push_swap 12 45 3 34 3 | ./checker_linux 12 45 3 34 3
./push_swap 12 45 a 34 3 | ./checker_linux 12 45 3 34 3
valgrind ./push_swap 12 45 a 34 3 | ./checker_linux 12 45 3 34 3
* more complicated stream
shuf -i 0-9999 -n 500 > args.txt ; ./push_swap --bench $(cat args.txt) 2> bench.txt | ./checker_linux $(cat args.txt)
shuf -i 0-9999 -n 500 > args.txt ; valgrind ./push_swap --bench $(cat args.txt) 2> bench.txt | ./checker_linux $(cat args.txt)
* numbers and arrays together
./push_swap "     12    5    " 45 '1' "  34 56" | ./checker_linux "     12    5    " 45 '1' "  34 56"
valgrind ./push_swap "     12    5    " 45 '1' "  34 56" | ./checker_linux "     12    5    " 45 '1' "  34 56"
./push_swap '12 34 56' '    34' "12 34 56" 45 | ./checker_linux '12 34 56' '    34' "12 34 56" 45
valgrind ./push_swap '12 34 56' '    34' "12 34 56" 45 | ./checker_linux '12 34 56' '    34' "12 34 56" 45
./push_swap "12 45 2" '  5 2 3  12' | ./checker_linux "12 45 2" '  5 2 3  12'
valgrind ./push_swap "12 45 2" '  5 2 3  12' | ./checker_linux "12 45 2" '  5 2 3  12'
* arrays
ARG="4 67 3 87 23"; valgrind ./push_swap --complex $ARG | ./checker_linux $ARG

./push_swap --complex 4 -34 12 -2000 -2147483648 | ./checker_linux 4 -34 12 -2000 -2147483648
valgrind ./push_swap --complex 4 -34 12 -2000 -2147483648 | ./checker_linux 4 -34 12 -2000 -2147483648
* complex z INT_MAX and INT_MIN

./push_swap --complex "4 -34 12         2147483647 -2000 -2147483648" -45 '23 -20 -47687' | ./checker_linux "4 -34 12         2147483647 -2000 -2147483648" -45 '23 -20 -47687'
